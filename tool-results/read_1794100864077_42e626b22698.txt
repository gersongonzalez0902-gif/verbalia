    65→/** Nivel de insignia según recompensa: bronce / plata / oro / platino. */
    66→function badgeTier(xpReward: number): { label: string; from: string; to: string } {
    67→  if (xpReward >= 150) return { label: "Platino", from: "oklch(0.85 0.06 240)", to: "oklch(0.95 0.03 240)" };
    68→  if (xpReward >= 100) return { label: "Oro", from: "oklch(0.75 0.15 85)", to: "oklch(0.88 0.12 95)" };
    69→  if (xpReward >= 50) return { label: "Plata", from: "oklch(0.62 0.05 250)", to: "oklch(0.82 0.03 250)" };
    70→  return { label: "Bronce", from: "oklch(0.55 0.09 55)", to: "oklch(0.72 0.09 65)" };
    71→}
