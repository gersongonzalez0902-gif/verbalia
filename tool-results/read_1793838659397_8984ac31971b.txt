   166→/** Comentarios de Tinta en la pantalla de resultados. Cada franja tiene un
   167→ * repertorio: la misma exactitud nunca cansa con la misma frase. */
   168→function pick<T>(pool: T[]): T {
   169→  return pool[Math.floor(Math.random() * pool.length)];
   170→}
