    64→/** Nivel de insignia según recompensa: bronce / plata / oro / platino. */
    65→function badgeTier(xpReward: number): { label: string; from: string; to: string } {
    66→  if (xpReward >= 150) return { label: "Platino", from: "oklch(0.85 0.06 240)", to: "oklch(0.95 0.03 240)" };
    67→  if (xpReward >= 100) return { label: "Oro", from: "oklch(0.75 0.15 85)", to: "oklch(0.88 0.12 95)" };
    68→  if (xpReward >= 50) return { label: "Plata", from: "oklch(0.62 0.05 250)", to: "oklch(0.82 0.03 250)" };
    69→  return { label: "Bronce", from: "oklch(0.55 0.09 55)", to: "oklch(0.72 0.09 65)" };
    70→}

   452→          </p>

   455→          {featured && <FeaturedBadgeCard badge={featured} onRemove={() => toggleFeatured(featured.id)} />}

   471→                    "group relative flex aspect-square items-center justify-center rounded-2xl border p-1.5 transition",

   565→                </AlertDialogAction>

   567→            </AlertDialogContent>

   570→
   571→        <Button variant="outline" className="mt-4 w-full font-bold" onClick={() => { sfx.click(); navigate({ name: "settings" }); }}>
   572→          Ajustes de temas, sonido y datos →

   574→      </main>
