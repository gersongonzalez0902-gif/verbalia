    15→import { GAMES } from "@/data/games";
    16→import { levelProgress } from "@/engine/xp";
    17→import { getIcon } from "@/components/game/icons";
    18→import { ScreenHeader } from "@/components/game/app-shell";
    19→import { Button } from "@/components/ui/button";

    25→
    26→export function MapScreen() {
    27→  const core = useGameStore((s) => s.core);
    28→  const navigate = useUiStore((s) => s.navigate);
    29→  const [openRegion, setOpenRegion] = useState<RegionId | null>(null);
    30→  const { level } = levelProgress(core.xp);
    31→
    32→  const regionProgress = (regionId: RegionId) => {
    33→    const region = REGION_BY_ID[regionId];
    34→    const days = Object.keys(core.stats.sessionsByDay);
    35→    const played = days.filter((d) => (core.stats.regionsByDay[d] ?? []).includes(regionId)).length;
    36→    // Progreso relativo: sesiones jugadas de esa región sobre 10 (sensación de avance).
    37→    return { played, pct: Math.min(1, played / 10) };
    38→  };
    39→
    40→  const regionAccuracy = (regionId: RegionId) => {
    41→    const region = REGION_BY_ID[regionId];
    42→    let answers = 0;
    43→    let correct = 0;
    44→    for (const g of region.games) {
    45→      const s = core.stats.byGame[g];
    46→      if (s) {
    47→        answers += s.answers;
    48→        correct += s.correct;
    49→      }
    50→    }
    51→    return answers > 0 ? Math.round((correct / answers) * 100) : null;
    52→  };
    53→
    54→  return (
    55→    <div>
    56→      <ScreenHeader title="El Archipiélago de las Palabras" subtitle="Ocho regiones, ocho caminos" backTo={{ name: "home" }} />
    57→
    58→      <main className="mx-auto w-full max-w-4xl px-4 pb-28">
    59→        {/* ===== Escena del mapa ===== */}
    60→        <div
    61→          className="relative mt-4 h-[68dvh] max-h-[640px] min-h-[440px] w-full overflow-hidden rounded-3xl border-2 border-primary/20 shadow-md sm:h-auto sm:aspect-[16/10] sm:max-h-none sm:min-h-0"
    62→          role="img"
    63→          aria-label="Mapa del Archipiélago de las Palabras con ocho regiones"
    64→        >
    65→          <IslandScene />
    66→
    67→          {/* Botones-isla superpuestos (accesibles y táctiles) */}
    68→          {REGIONS.map((r, index) => {
    69→            const shipped = r.games.length > 0;
    70→            const unlocked = shipped && level >= r.unlockLevel;
    71→            const prog = regionProgress(r.id);
    72→            const Icon = getIcon(r.icon);
    73→            return (
    74→              <button
    75→                key={r.id}
    76→                onClick={() => { if (unlocked) { sfx.buoy(); setOpenRegion(r.id); } }}
    77→                disabled={!unlocked}
    78→                aria-label={`${r.name}. ${unlocked ? `${prog.played} sesiones jugadas. Tocar para abrir la región.` : shipped ? `Se desbloquea al alcanzar el nivel ${r.unlockLevel}.` : `Región en desarrollo: llegará en la Etapa ${r.stage}.`}`}
    79→                className={cn(
    80→                  "group absolute z-10 flex -translate-x-1/2 -translate-y-1/2 flex-col items-center gap-1 rounded-2xl p-2 transition-all duration-200",
    81→                  unlocked ? "hover:scale-110 focus-visible:scale-110 focus-visible:outline-2 focus-visible:outline-primary" : "cursor-default",
    82→                )}
    83→                style={{ left: `${r.map.x}%`, top: `${r.map.y}%` }}
    84→              >
    85→                {/* Halo regional tras la isla desbloqueada */}
    86→                {unlocked && (
    87→                  <span
    88→                    aria-hidden
    89→                    className="anim-glow absolute left-1/2 top-1/2 -z-10 h-20 w-20 -translate-x-1/2 -translate-y-1/2 rounded-full"
    90→                    style={{ background: `color-mix(in oklch, var(${r.colorVar}) 32%, transparent)`, filter: "blur(18px)" }}
    91→                  />
    92→                )}
    93→                <span
    94→                  className={cn(
    95→                    "gloss anim-bob relative flex h-16 w-16 items-center justify-center rounded-full border-4 text-white sm:h-[4.5rem] sm:w-[4.5rem]",
    96→                    unlocked ? "border-white/90" : "border-white/50",
    97→                  )}
    98→                  style={{
    99→                    background: `linear-gradient(150deg, color-mix(in oklch, var(${r.colorVar}) ${unlocked ? 74 : 52}%, ${unlocked ? "white" : "#5a5a6e"} ${unlocked ? 18 : 26}%), var(${r.colorVar}) ${unlocked ? 55 : 60}%, color-mix(in oklch, var(${r.colorVar}) ${unlocked ? 80 : 62}%, #000 ${unlocked ? 18 : 38}%))`,
   100→                    boxShadow: unlocked
   101→                      ? `0 4px 0 0 color-mix(in oklch, var(${r.colorVar}) 50%, #000 30%), 0 8px 16px -8px color-mix(in oklch, var(${r.colorVar}) 55%, transparent)`
   102→                      : `0 3px 0 0 color-mix(in oklch, var(${r.colorVar}) 35%, #000 45%)`,
   103→                    animationDelay: `${index * 0.7}s`,
   104→                  }}
   105→                >
   106→                  <Icon className="h-6 w-6 sm:h-7 sm:w-7" aria-hidden />
   107→                  {!unlocked && (
   108→                    <span className="absolute -bottom-1.5 -right-2 flex items-center gap-0.5 rounded-full bg-card px-1 py-0.5 text-[9px] font-black shadow" aria-hidden>
   109→                      <Lock className="h-3 w-3" />
   110→                      <span className="numeric text-foreground">{r.unlockLevel}</span>
   111→                    </span>
   112→                  )}
   113→                </span>
   114→                <span
   115→                  className={cn(
   116→                    "font-display max-w-28 rounded-full px-2 py-0.5 text-center text-[10px] font-bold leading-tight shadow-sm sm:text-xs",
   117→                    unlocked ? "bg-card text-foreground" : "bg-muted text-muted-foreground",
   118→                  )}
   119→                >
   120→                  {r.name.replace(" de la ", " · ").replace(" del ", " · ")}
   121→                </span>

   124→          })}

   126→          {/* Brújula decorativa */}

   128→            <span className="text-lg">🧭</span>
   129→          </div>
   130→        </div>
   131→
   132→        {/* ===== Leyenda de estados ===== */}
   133→        <div className="mt-4 flex flex-wrap items-center justify-center gap-x-5 gap-y-2 text-xs text-muted-foreground">
   134→          <span className="flex items-center gap-1.5">
   135→            <span className="h-3 w-3 rounded-full bg-primary" aria-hidden /> Ocho islas jugables
   136→          </span>
   137→          <span className="flex items-center gap-1.5">
   138→            <span className="h-3 w-3 rounded-full" style={{ background: "var(--region-forja)" }} aria-hidden /> Etapa 4 · la Forja del Escritor
   139→          </span>
   140→          <span className="flex items-center gap-1.5">
   141→            <Lock className="h-3.5 w-3.5" aria-hidden /> Sube de nivel para desbloquear
   142→          </span>
   143→        </div>
   144→
   145→        {/* ===== Lista accesible de regiones ===== */}
   146→        <section className="mt-6" aria-label="Lista de regiones">
   147→          <h2 className="font-display mb-2.5 text-lg font-bold">Vista de lista</h2>
   148→          <ul className="grid gap-2.5 sm:grid-cols-2">

   150→              const shipped = r.games.length > 0;
   151→              const unlocked = shipped && level >= r.unlockLevel;

   173→                      <Icon className="h-5 w-5" />

   180→                          : !shipped
   181→                            ? `Llega en la Etapa ${r.stage}`
   182→                            : `Nivel ${r.unlockLevel} necesario`}
   183→                      </span>
   184→                    </span>
   185→                    {unlocked && <Sparkles className="h-4 w-4 shrink-0 text-primary" aria-hidden />}
   186→                  </button>
   187→                </li>

   191→        </section>
   192→      </main>
   193→

   195→      <RegionSheet

   213→/* ============ Ficha de región (hoja lateral) ============ */
   214→
   215→function RegionSheet({
   216→  regionId,
   217→  onClose,
   218→  onPlayMissions,
   219→  onTrainGame,
   220→  progressOf,
   221→  accuracyOf,
   222→}: {
   223→  regionId: RegionId | null;
   224→  onClose: () => void;
   225→  onPlayMissions: (regionId: RegionId) => void;
   226→  onTrainGame: (gameId: GameId) => void;
   227→  progressOf: (regionId: RegionId) => { played: number; pct: number };
   228→  accuracyOf: (regionId: RegionId) => number | null;
   229→}) {
   230→  const region = regionId ? REGION_BY_ID[regionId] : null;
   231→  const byGame = useGameStore((s) => s.core.stats.byGame);
   232→  if (!region) return null;
   233→  const prog = progressOf(region.id);
   234→  const accuracy = accuracyOf(region.id);
   235→  const skillLabel = SKILL_LABELS[region.skill] ?? region.skill;
   236→
   237→  return (
   238→    <Sheet open={regionId !== null} onOpenChange={(v) => !v && onClose()}>
   239→      <SheetContent side="bottom" className="safe-bottom max-h-[85dvh] overflow-y-auto rounded-t-3xl sm:max-w-lg sm:mx-auto sm:rounded-3xl sm:bottom-auto sm:top-1/2 sm:-translate-y-1/2">
   240→        <SheetHeader className="pb-0 text-left">
   241→          <div className="flex items-center gap-3.5">
   242→            <span
   243→              className="gloss flex h-13 w-13 shrink-0 items-center justify-center rounded-2xl p-3 text-white"
   244→              style={{
   245→                background: `linear-gradient(150deg, color-mix(in oklch, var(${region.colorVar}) 76%, white 18%), var(${region.colorVar}) 58%, color-mix(in oklch, var(${region.colorVar}) 82%, #000 16%))`,
   246→                boxShadow: `0 3px 0 0 color-mix(in oklch, var(${region.colorVar}) 52%, #000 30%)`,
   247→              }}
   248→              aria-hidden
   249→            >
   250→              {createElement(getIcon(region.icon), { className: "h-6 w-6" })}
   251→            </span>
   252→            <div>
   253→              <SheetTitle className="font-display text-xl font-bold leading-tight">{region.name}</SheetTitle>
   254→              <SheetDescription>{region.subtitle}</SheetDescription>
   255→            </div>
   256→          </div>
   257→        </SheetHeader>
   258→
   259→        <div className="space-y-5 px-4 pb-6 pt-3">
   260→          <p className="text-sm leading-relaxed text-muted-foreground">{region.description}</p>
   261→
   262→          <p className="rounded-2xl bg-secondary px-4 py-2.5 text-xs font-semibold text-secondary-foreground">
   263→            Habilidad que entrena: {skillLabel}
   264→          </p>
   265→
   266→          <div className="flex items-center gap-4">
   267→            <ProgressRing value={prog.pct} size={86} stroke={8} color={`var(${region.colorVar})`}>
   268→              <span className="numeric font-display text-lg font-bold">{Math.round(prog.pct * 100)}%</span>
   269→            </ProgressRing>
   270→            <div className="text-sm">
   271→              <p className="font-bold">{prog.played} sesiones jugadas</p>
   272→              <p className="text-muted-foreground">
   273→                {accuracy !== null ? `Exactitud media: ${accuracy}%` : "Aún sin datos en esta región"}
   274→              </p>
   275→            </div>
   276→          </div>
   277→
   278→          <Button
   279→            size="lg"
   280→            className="w-full text-base"
   281→            style={{
   282→              background: `var(${region.colorVar})`,
   283→              color: "white",
   284→              ["--btn-edge" as string]: `color-mix(in oklch, var(${region.colorVar}) 50%, #000 32%)`,
   285→            }}
   286→            onClick={() => { sfx.whoosh(); onPlayMissions(region.id); }}
   287→          >
   288→            <Play className="mr-2 h-5 w-5" aria-hidden />
   289→            Misión de la región (3-5 min)
   290→          </Button>
   291→
   292→          <div>
   293→            <p className="mb-2 text-xs font-bold uppercase tracking-widest text-muted-foreground">
   294→              Entrenamientos
   295→            </p>
   296→            <div className="space-y-2">
   297→              {region.games.map((gid) => {
   298→                const g = GAMES[gid];
   299→                if (!g) return null;
   300→                const Icon = getIcon(g.icon);
   301→                const stats = byGame[gid];
   302→                return (
   303→                  <button
   304→                    key={gid}
   305→                    onClick={() => { sfx.whoosh(); onTrainGame(gid); }}
   306→                    className="tap-3d flex w-full items-center gap-3 rounded-2xl border bg-card p-3.5 text-left"
   307→                  >
   308→                    <span className="chip-icon flex h-10 w-10 shrink-0 items-center justify-center rounded-xl">
   309→                      <Icon className="h-5 w-5" aria-hidden />
   310→                    </span>
   311→                    <span className="min-w-0 flex-1">
   312→                      <span className="block text-sm font-bold">{g.name}</span>
   313→                      <span className="block truncate text-xs text-muted-foreground">{g.tagline}</span>
   314→                    </span>
   315→                    <span className="shrink-0 text-right text-[11px] text-muted-foreground">
   316→                      {stats ? `${stats.plays} partidas` : "nuevo"}
   317→                    </span>
   318→                  </button>
   319→                );
   320→              })}
   321→            </div>
   322→          </div>
   323→        </div>
   324→      </SheetContent>
   325→    </Sheet>
   326→  );
   327→}
