     1→     1→/**
     2→     2→ * Motor de audio de VERBALIA · v2
     3→     3→ * ---------------------------------------------------------------
     4→     4→ * Todo el sonido del juego se sintetiza aquí con WebAudio:
     5→     5→ * sin ficheros, sin descargas, sin red.
     6→     6→ *
     7→     7→ * 1) SFX: ~30 efectos de interacción (acierto, fallo, niveles,
     8→     8→ *    monedas de XP, naipes, etc.) con variantes para que nunca
     9→     9→ *    suenen igual dos veces seguidas.
    10→    10→ * 2) MÚSICA AMBIENTE GENERATIVA: cada escena/ región tiene su
    11→    11→ *    propia paleta musical (escala, tempo, timbres, textura de
    12→    12→ *    ruido) compuesta en tiempo real con un secuenciador por
    13→    13→ *    adelantado (lookahead). Al cambiar de región la música
    14→    14→ *    hace fundido cruzado.
    15→    15→ * 3) EL GRAN RELOJ EN CONTRATIEMPO: una capa de tic-tac que
    16→    16→ *    suena DESFASADA respecto al pulso musical (en los
    17→    17→ *    contratiempos, como pidió el jugador). En reposo es una
    18→    18→ *    presencia mínima y sugerente; en los juegos cronometrados
    19→    19→ *    `setTension()` la enciende, la acelera y le sube el tono
    20→    20→ *    hasta el clímax de los últimos segundos.
    21→    21→ *
    22→    22→ * El AudioContext se desbloquea con el primer gesto del usuario
    23→    23→ * (política de navegadores). Si el jugador silencia el juego,
    24→    24→ * el contexto se suspende y todo queda en silencio absoluto.
    25→    25→ */
    26→    26→
    27→    27→/* ============================== ajustes ============================== */
    28→    28→

    30→    30→let master: GainNode | null = null;

    32→    32→let sfxBus: GainNode | null = null;

    34→    34→let noiseBus: GainNode | null = null;
    35→    35→
    36→    36→let enabled = true;
    37→    37→let musicVolume = 0.55;
    38→    38→let sfxVolume = 0.9;
    39→    39→
    40→    40→let unlocked = false;
    41→    41→let unlockListeners: (() => void) | null = null;

    44→    44→let noiseBuffer: AudioBuffer | null = null;
    45→    45→
    46→    46→function ensureCtx(): AudioContext | null {
    47→    47→  if (typeof window === "undefined") return null;
    48→    48→  try {
    49→    49→    if (!ctx) {
    50→    50→      const AudioCtor =
    51→    51→        window.AudioContext ??
    52→    52→        (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
    53→    53→      if (!AudioCtor) return null;
    54→    54→      ctx = new AudioCtor();
    55→    55→      master = ctx.createGain();
    56→    56→      master.gain.value = enabled ? 0.9 : 0;
    57→    57→      master.connect(ctx.destination);
    58→    58→
    59→    59→      musicBus = ctx.createGain();
    60→    60→      musicBus.gain.value = musicVolume;
    61→    61→      musicBus.connect(master);
    62→    62→
    63→    63→      sfxBus = ctx.createGain();
    64→    64→      sfxBus.gain.value = sfxVolume;
    65→    65→      sfxBus.connect(master);
    66→    66→
    67→    67→      clockBus = ctx.createGain();
    68→    68→      clockBus.gain.value = 0.5;
    69→    69→      clockBus.connect(master);
    70→    70→
    71→    71→      noiseBus = ctx.createGain();
    72→    72→      noiseBus.gain.value = 1;
    73→    73→      noiseBus.connect(musicBus);
    74→    74→
    75→    75→      const len = ctx.sampleRate * 2;
    76→    76→      noiseBuffer = ctx.createBuffer(1, len, ctx.sampleRate);
    77→    77→      const data = noiseBuffer.getChannelData(0);
    78→    78→      for (let i = 0; i < len; i++) data[i] = Math.random() * 2 - 1;
    79→    79→    }
    80→    80→    if (ctx.state === "suspended" && enabled) void ctx.resume();
    81→    81→    return ctx;
    82→    82→  } catch {
    83→    83→    return null;
    84→    84→  }
    85→    85→}

    87→    87→/** Desbloquea el audio en el primer gesto (tap, teclado, scroll táctil). */
    88→    88→export function initAudioUnlock(): void {
    89→    89→  if (typeof window === "undefined" || unlockListeners) return;
    90→    90→  const unlock = () => {
    91→    91→    ensureCtx();
    92→    92→    unlocked = true;
    93→    93→    if (currentScene) startScene(currentScene); // la música arranca al desbloquear
    94→    94→    detach();
    95→    95→  };
    96→    96→  const detach = () => {
    97→    97→    window.removeEventListener("pointerdown", unlock);
    98→    98→    window.removeEventListener("keydown", unlock);
    99→    99→    window.removeEventListener("touchstart", unlock);
   100→   100→    unlockListeners = null;
   101→   101→  };
   102→   102→  window.addEventListener("pointerdown", unlock, { once: false, passive: true });
   103→   103→  window.addEventListener("keydown", unlock, { once: false });
   104→   104→  window.addEventListener("touchstart", unlock, { once: false, passive: true });
   105→   105→  unlockListeners = detach;
   106→   106→}

   108→   108→export function setSoundEnabled(value: boolean) {
   109→   109→  enabled = value;
   110→   110→  const ac = ctx ?? null;
   111→   111→  if (ac && master) master.gain.setTargetAtTime(value ? 0.9 : 0, ac.currentTime, 0.05);
   112→   112→  if (!value && ac && ac.state === "running") void ac.suspend();
   113→   113→  if (value && ac && ac.state === "suspended") {
   114→   114→    void ac.resume();
   115→   115→    unlocked = true;
   116→   116→  }

   135→   135→function tone(
   136→   136→  freq: number,
   137→   137→  start: number,
   138→   138→  duration: number,
   139→   139→  volume = 0.09,
   140→   140→  type: OscillatorType = "sine",
   141→   141→  bus: GainNode | null = null,
   142→   142→  glideTo?: number,
   143→   143→) {
   144→   144→  const ac = ensureCtx();
   145→   145→  if (!ac || !enabled) return;
   146→   146→  const osc = ac.createOscillator();
   147→   147→  const gain = ac.createGain();
   148→   148→  osc.type = type;
   149→   149→  const t = ac.currentTime + start;
   150→   150→  osc.frequency.setValueAtTime(freq, t);
   151→   151→  if (glideTo) osc.frequency.exponentialRampToValueAtTime(Math.max(20, glideTo), t + duration);
   152→   152→  gain.gain.setValueAtTime(0.0001, t);
   153→   153→  gain.gain.exponentialRampToValueAtTime(Math.max(volume, 0.0002), t + 0.012);
   154→   154→  gain.gain.exponentialRampToValueAtTime(0.0001, t + duration);
   155→   155→  osc.connect(gain).connect(bus ?? sfxBus ?? ac.destination);
   156→   156→  osc.start(t);
   157→   157→  osc.stop(t + duration + 0.06);
   158→   158→}

   160→   160→/** Campanada tipo campana con parciales inarmónicos (para bells y éxito). */
   161→   161→function bell(freq: number, start: number, duration = 0.9, volume = 0.08) {
   162→   162→  const ac = ensureCtx();
   163→   163→  if (!ac || !enabled) return;
   164→   164→  const partials = [1, 2.76, 5.4];
   165→   165→  partials.forEach((p, i) => {
   166→   166→    tone(freq * p, start, duration / (1 + i * 0.4), volume / (i + 1.5), "sine");
   167→   167→  });
   168→   168→}

   170→   170→/** Ráfaga de ruido filtrada (clics, olas, viento, aplausos). */
   171→   171→function noiseBurst(
   172→   172→  start: number,
   173→   173→  duration: number,
   174→   174→  filterFreq: number,
   175→   175→  volume = 0.06,
   176→   176→  type: BiquadFilterType = "bandpass",
   177→   177→  q = 1,
   178→   178→  bus: GainNode | null = null,
   179→   179→) {
   180→   180→  const ac = ensureCtx();
   181→   181→  if (!ac || !enabled || !noiseBuffer) return;
   182→   182→  const src = ac.createBufferSource();
   183→   183→  src.buffer = noiseBuffer;
   184→   184→  src.loop = true;
   185→   185→  const filter = ac.createBiquadFilter();
   186→   186→  filter.type = type;
   187→   187→  filter.frequency.value = filterFreq;
   188→   188→  filter.Q.value = q;
   189→   189→  const gain = ac.createGain();
   190→   190→  const t = ac.currentTime + start;
   191→   191→  gain.gain.setValueAtTime(0.0001, t);
   192→   192→  gain.gain.exponentialRampToValueAtTime(volume, t + Math.min(0.05, duration / 3));
   193→   193→  gain.gain.exponentialRampToValueAtTime(0.0001, t + duration);
   194→   194→  src.connect(filter).connect(gain).connect(bus ?? sfxBus ?? ac.destination);
   195→   195→  src.start(t, Math.random() * 1.5);
   196→   196→  src.stop(t + duration + 0.08);
   197→   197→}

   376→   376→/** Funda de entrada a la escena: acorde inicial y textura. */
   377→   377→function startScene(sceneId: string) {
   378→   378→  const ac = ensureCtx();
   379→   379→  if (!ac || !enabled) return;
   380→   380→  const scene = SCENES[sceneId] ?? SCENES.home;
   381→   381→  currentScene = sceneId;
   382→   382→  stopTexture();
   383→   383→  startTexture(scene);
   384→   384→  // Acorde de bienvenida
   385→   385→  scene.chords[0].forEach((d) => padNote(semitone(scene.root, scene.scale[d]), 0.02, 6, 0.05));
   386→   386→  if (!schedulerId) {
   387→   387→    nextBeatTime = ac.currentTime + 0.1;
   388→   388→    beatIndex = 0;
   389→   389→    barIndex = 0;
   390→   390→    schedulerId = setInterval(scheduleAhead, 240);
   391→   391→  }
   392→   392→}

   396→   396→  if (!sceneId || sceneId === currentScene) return;
   397→   397→  if (!ctx || !unlocked) {
   398→   398→    currentScene = sceneId; // se aplicará al desbloquear
   399→   399→    return;
   400→   400→  }
   401→   401→  startScene(sceneId);
   402→   402→}

   404→   404→function stopTexture() {
   405→   405→  if (textureNodes && ctx) {
   406→   406→    try {
   407→   407→      textureNodes.gain.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.4);
   408→   408→      const src = textureNodes.src;
   409→   409→      setTimeout(() => {
   410→   410→        try {
   411→   411→          src.stop();
   412→   412→        } catch {
   413→   413→          /* ya parado */
   414→   414→        }
   415→   415→      }, 1500);
   416→   416→    } catch {
   417→   417→      /* ignorar */
   418→   418→    }
   419→   419→    textureNodes = null;
   420→   420→  }

   423→   423→function startTexture(scene: SceneMusic) {
   424→   424→  const ac = ctx;
   425→   425→  if (!ac || !noiseBuffer) return;
   426→   426→  const src = ac.createBufferSource();
   427→   427→  src.buffer = noiseBuffer;
   428→   428→  src.loop = true;
   429→   429→  const filter = ac.createBiquadFilter();
   430→   430→  const gain = ac.createGain();
   431→   431→  gain.gain.value = 0.0001;
   432→   432→  const t = ac.currentTime;
   433→   433→  let lfo: OscillatorNode | null = null;
   434→   434→  let lfoGain: GainNode | null = null;
   435→   435→
   436→   436→  switch (scene.texture) {
   437→   437→    case "wind":
   438→   438→      filter.type = "lowpass";
   439→   439→      filter.frequency.value = 420;
   440→   440→      // ráfagas lentas
   441→   441→      lfo = ac.createOscillator();
   442→   442→      lfo.frequency.value = 0.09;
   443→   443→      lfoGain = ac.createGain();
   444→   444→      lfoGain.gain.value = scene.textureLevel * 0.7;
   445→   445→      lfo.connect(lfoGain).connect(gain.gain);
   446→   446→      gain.gain.value = scene.textureLevel;
   447→   447→      break;
   448→   448→    case "waves":
   449→   449→      filter.type = "lowpass";
   450→   450→      filter.frequency.value = 300;
   451→   451→      lfo = ac.createOscillator();
   452→   452→      lfo.frequency.value = 0.07; // marea muy lenta
   453→   453→      lfoGain = ac.createGain();
   454→   454→      lfoGain.gain.value = scene.textureLevel * 0.8;
   455→   455→      lfo.connect(lfoGain).connect(gain.gain);
   456→   456→      gain.gain.value = scene.textureLevel;
   457→   457→      break;
   458→   458→    case "hum":
   459→   459→      filter.type = "lowpass";
   460→   460→      filter.frequency.value = 180;
   461→   461→      gain.gain.setTargetAtTime(scene.textureLevel, t, 1.2);
   462→   462→      break;
   463→   463→    case "paper":
   464→   464→      filter.type = "bandpass";
   465→   465→      filter.frequency.value = 4200;
   466→   466→      filter.Q.value = 0.4;
   467→   467→      gain.gain.setTargetAtTime(scene.textureLevel * 0.5, t, 1.5);
   468→   468→      break;
   469→   469→    case "crackle":
   470→   470→      filter.type = "bandpass";
   471→   471→      filter.frequency.value = 2600;
   472→   472→      filter.Q.value = 2.5;
   473→   473→      lfo = ac.createOscillator();
   474→   474→      lfo.frequency.value = 1.7;
   475→   475→      lfoGain = ac.createGain();
   476→   476→      lfoGain.gain.value = scene.textureLevel;
   477→   477→      lfo.connect(lfoGain).connect(gain.gain);
   478→   478→      gain.gain.value = scene.textureLevel * 0.3;
   479→   479→      break;
   480→   480→    case "air":
   481→   481→    default:
   482→   482→      filter.type = "lowpass";
   483→   483→      filter.frequency.value = 900;
   484→   484→      gain.gain.setTargetAtTime(scene.textureLevel, t, 1.5);
   485→   485→      break;
   486→   486→  }
   487→   487→  src.connect(filter).connect(gain).connect(noiseBus ?? master ?? ac.destination);
   488→   488→  if (lfo && lfoGain) lfo.start();
   489→   489→  src.start();
   490→   490→  textureNodes = { src, gain };
   491→   491→}

   493→   493→/** Colchón armónico suave (pad) con ataque largo y filtro móvil. */
   494→   494→function padNote(freq: number, start: number, duration: number, volume: number) {
   495→   495→  const ac = ctx;
   496→   496→  if (!ac || !enabled) return;
   497→   497→  const t = ac.currentTime + start;
   498→   498→  const gain = ac.createGain();
   499→   499→  gain.gain.setValueAtTime(0.0001, t);
   500→   500→  gain.gain.linearRampToValueAtTime(volume, t + duration * 0.35);
   501→   501→  gain.gain.setTargetAtTime(0.0001, t + duration * 0.6, 1.2);
   502→   502→  const filter = ac.createBiquadFilter();
   503→   503→  filter.type = "lowpass";
   504→   504→  filter.frequency.setValueAtTime(freq * 1.4, t);
   505→   505→  filter.frequency.linearRampToValueAtTime(freq * 0.9, t + duration);
   506→   506→  [0, 4, -5].forEach((detune) => {
   507→   507→    const osc = ac.createOscillator();
   508→   508→    osc.type = "sawtooth";
   509→   509→    osc.frequency.value = freq;
   510→   510→    osc.detune.value = detune * 3;
   511→   511→    osc.connect(filter);
   512→   512→    osc.start(t);
   513→   513→    osc.stop(t + duration + 1.8);
   514→   514→  });
   515→   515→  filter.connect(gain).connect(musicBus ?? ac.destination);
   516→   516→}

   518→   518→/** Nota melódica breve (arpegio/pltuck) con carácter de la escena. */
   519→   519→function pluck(freq: number, start: number, volume: number, type: OscillatorType) {
   520→   520→  const ac = ctx;
   521→   521→  if (!ac || !enabled) return;
   522→   522→  const t = ac.currentTime + start;
   523→   523→  const osc = ac.createOscillator();
   524→   524→  const gain = ac.createGain();
   525→   525→  osc.type = type;
   526→   526→  osc.frequency.value = freq;
   527→   527→  gain.gain.setValueAtTime(0.0001, t);
   528→   528→  gain.gain.exponentialRampToValueAtTime(volume, t + 0.02);
   529→   529→  gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.55);
   530→   530→  osc.connect(gain).connect(musicBus ?? ac.destination);
   531→   531→  osc.start(t);
   532→   532→  osc.stop(t + 0.65);
   533→   533→}

   535→   535→/** El tic del Gran Reloj: madera seca y corta. */
   536→   536→function clockTick(high: boolean, start: number, volume: number) {
   537→   537→  const ac = ctx;
   538→   538→  if (!ac || !enabled) return;
   539→   539→  const t = ac.currentTime + start;
   540→   540→  const osc = ac.createOscillator();
   541→   541→  const gain = ac.createGain();
   542→   542→  osc.type = "square";
   543→   543→  osc.frequency.setValueAtTime(high ? 1980 : 1560, t);
   544→   544→  osc.frequency.exponentialRampToValueAtTime(high ? 1500 : 1180, t + 0.03);
   545→   545→  gain.gain.setValueAtTime(0.0001, t);
   546→   546→  gain.gain.exponentialRampToValueAtTime(volume, t + 0.004);
   547→   547→  gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.05);
   548→   548→  const filter = ac.createBiquadFilter();
   549→   549→  filter.type = "bandpass";
   550→   550→  filter.frequency.value = 1900;
   551→   551→  filter.Q.value = 4;
   552→   552→  osc.connect(filter).connect(gain).connect(clockBus ?? ac.destination);
   553→   553→  osc.start(t);
   554→   554→  osc.stop(t + 0.07);
   555→   555→  // chasquido de mecánica
   556→   556→  noiseBurst(start, 0.018, 3400, volume * 0.4, "bandpass", 5, clockBus);
   557→   557→}

   559→   559→function scheduleAhead() {
   560→   560→  const ac = ctx;
   561→   561→  const scene = SCENES[currentScene ?? "home"] ?? SCENES.home;
   562→   562→  if (!ac || !enabled) return;
   563→   563→  const beatDur = 60 / scene.bpm;
   564→   564→  while (nextBeatTime < ac.currentTime + 0.6) {
   565→   565→    const start = Math.max(0, nextBeatTime - ac.currentTime);
   566→   566→    const beatInBar = beatIndex % 4;
   567→   567→
   568→   568→    /* -- pulso: colchón en cada compás, bajo en el 1 -- */
   569→   569→    if (beatInBar === 0) {
   570→   570→      const chord = scene.chords[barIndex % scene.chords.length];
   571→   571→      chord.forEach((d) =>
   572→   572→        padNote(semitone(scene.root * 2, scene.scale[d]), start, beatDur * 4.2, 0.035),
   573→   573→      );
   574→   574→      if (scene.bass) padNote(scene.root / 2, start, beatDur * 4, 0.05);
   575→   575→    }
   576→   576→
   577→   577→    /* -- arpegios en los contratiempos (nunca en el pulso fuerte) -- */
   578→   578→    const offbeat = start + beatDur * 0.5;
   579→   579→    if (Math.random() < scene.pluck) {
   580→   580→      const deg = scene.scale[Math.floor(Math.random() * scene.scale.length)];
   581→   581→      const octave = Math.random() < 0.3 ? 24 : 12;
   582→   582→      pluck(semitone(scene.root, deg + octave), offbeat, 0.055, scene.pluckType);
   583→   583→    }
   584→   584→    if (beatInBar === 2 && Math.random() < scene.pluck * 0.7) {
   585→   585→      const deg = scene.scale[Math.floor(Math.random() * scene.scale.length)];
   586→   586→      pluck(semitone(scene.root, deg + 12), start + beatDur * 0.25, 0.04, "sine");
   587→   587→    }
   588→   588→
   589→   589→    /* -- EL GRAN RELOJ EN CONTRATIEMPO --
   590→   590→       tic-tac entre pulsos: en reposo muy sutil; con tensión se
   591→   591→       vuelve insistente (cada contratiempo) y sube de tono. */
   592→   592→    const clockBase = clockOn ? 0.055 : 0.018;
   593→   593→    const clockVol = clockBase * (0.35 + tension * 1.65) * (beatInBar % 2 === 0 ? 1 : 0.8);
   594→   594→    clockTick(clockTickToggle % 2 === 0, offbeat, clockVol);
   595→   595→    clockTickToggle++;
   596→   596→    if (tension > 0.55) {
   597→   597→      // doble tic nervioso en el segundo contratiempo del pulso
   598→   598→      clockTick(clockTickToggle % 2 === 0, offbeat + beatDur * 0.25, clockVol * 0.6);
   599→   599→    }
   600→   600→
   601→   601→    beatIndex++;
   602→   602→    if (beatInBar === 3) barIndex++;
   603→   603→    nextBeatTime += beatDur;
   604→   604→  }
   605→   605→}
