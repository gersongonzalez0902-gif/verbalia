     1→/**
     2→ * Motor de audio de VERBALIA · v2
     3→ * ---------------------------------------------------------------
     4→ * Todo el sonido del juego se sintetiza aquí con WebAudio:
     5→ * sin ficheros, sin descargas, sin red.
     6→ *
     7→ * 1) SFX: ~30 efectos de interacción (acierto, fallo, niveles,
     8→ *    monedas de XP, naipes, etc.) con variantes para que nunca
     9→ *    suenen igual dos veces seguidas.
    10→ * 2) MÚSICA AMBIENTE GENERATIVA: cada escena/ región tiene su
    11→ *    propia paleta musical (escala, tempo, timbres, textura de
    12→ *    ruido) compuesta en tiempo real con un secuenciador por
    13→ *    adelantado (lookahead). Al cambiar de región la música
    14→ *    hace fundido cruzado.
    15→ * 3) EL GRAN RELOJ EN CONTRATIEMPO: una capa de tic-tac que
    16→ *    suena DESFASADA respecto al pulso musical (en los
    17→ *    contratiempos, como pidió el jugador). En reposo es una
    18→ *    presencia mínima y sugerente; en los juegos cronometrados
    19→ *    `setTension()` la enciende, la acelera y le sube el tono
    20→ *    hasta el clímax de los últimos segundos.
    21→ *
    22→ * El AudioContext se desbloquea con el primer gesto del usuario
    23→ * (política de navegadores). Si el jugador silencia el juego,
    24→ * el contexto se suspende y todo queda en silencio absoluto.
    25→ */
    26→
    27→/* ============================== ajustes ============================== */
    28→
    29→let ctx: AudioContext | null = null;
    30→let master: GainNode | null = null;
    31→let musicBus: GainNode | null = null;
    32→let sfxBus: GainNode | null = null;
    33→let clockBus: GainNode | null = null;
    34→let noiseBus: GainNode | null = null;
    35→
    36→let enabled = true;
    37→let musicVolume = 0.55;
    38→let sfxVolume = 0.9;
    39→
    40→let unlocked = false;
    41→let unlockListeners: (() => void) | null = null;
    42→
    43→/* Ruido compartido (buffer de 2 s reutilizado por todas las texturas) */
    44→let noiseBuffer: AudioBuffer | null = null;
    45→
    46→function ensureCtx(): AudioContext | null {
    47→  if (typeof window === "undefined") return null;
    48→  try {
    49→    if (!ctx) {
    50→      const AudioCtor =
    51→        window.AudioContext ??
    52→        (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
    53→      if (!AudioCtor) return null;
    54→      ctx = new AudioCtor();
    55→      master = ctx.createGain();
    56→      master.gain.value = enabled ? 0.9 : 0;
    57→      master.connect(ctx.destination);
    58→
    59→      musicBus = ctx.createGain();
    60→      musicBus.gain.value = musicVolume;
    61→      musicBus.connect(master);
    62→
    63→      sfxBus = ctx.createGain();
    64→      sfxBus.gain.value = sfxVolume;
    65→      sfxBus.connect(master);
    66→
    67→      clockBus = ctx.createGain();
    68→      clockBus.gain.value = 0.5;
    69→      clockBus.connect(master);
    70→
    71→      noiseBus = ctx.createGain();
    72→      noiseBus.gain.value = 1;
    73→      noiseBus.connect(musicBus);
    74→
    75→      const len = ctx.sampleRate * 2;
    76→      noiseBuffer = ctx.createBuffer(1, len, ctx.sampleRate);
    77→      const data = noiseBuffer.getChannelData(0);
    78→      for (let i = 0; i < len; i++) data[i] = Math.random() * 2 - 1;
    79→    }
    80→    if (ctx.state === "suspended" && enabled) void ctx.resume();
    81→    return ctx;
    82→  } catch {
    83→    return null;
    84→  }
    85→}
    86→
    87→/** Desbloquea el audio en el primer gesto (tap, teclado, scroll táctil). */
    88→export function initAudioUnlock(): void {
    89→  if (typeof window === "undefined" || unlockListeners) return;
    90→  const unlock = () => {
    91→    ensureCtx();
    92→    unlocked = true;
    93→    if (currentScene) startScene(currentScene); // la música arranca al desbloquear
    94→    detach();
    95→  };
    96→  const detach = () => {
    97→    window.removeEventListener("pointerdown", unlock);
    98→    window.removeEventListener("keydown", unlock);
    99→    window.removeEventListener("touchstart", unlock);
   100→    unlockListeners = null;
   101→  };
   102→  window.addEventListener("pointerdown", unlock, { once: false, passive: true });
   103→  window.addEventListener("keydown", unlock, { once: false });
   104→  window.addEventListener("touchstart", unlock, { once: false, passive: true });
   105→  unlockListeners = detach;
   106→}

   108→export function setSoundEnabled(value: boolean) {
   109→  enabled = value;
   110→  const ac = ctx ?? null;
   111→  if (ac && master) master.gain.setTargetAtTime(value ? 0.9 : 0, ac.currentTime, 0.05);
   112→  if (!value && ac && ac.state === "running") void ac.suspend();
   113→  if (value && ac && ac.state === "suspended") {
   114→    void ac.resume();
   115→    unlocked = true;
   116→  }
   117→}

   119→export function setMusicVolume(v: number) {
   120→  musicVolume = Math.min(1, Math.max(0, v));
   121→  if (ctx && musicBus) musicBus.gain.setTargetAtTime(musicVolume, ctx.currentTime, 0.08);
   122→}

   124→export function setSfxVolume(v: number) {
   125→  sfxVolume = Math.min(1, Math.max(0, v));
   126→  if (ctx && sfxBus) sfxBus.gain.setTargetAtTime(sfxVolume, ctx.currentTime, 0.05);

   130→  return unlocked && !!ctx && ctx.state === "running";

   133→/* ============================== voces base ============================== */
   134→
   135→function tone(
   136→  freq: number,
   137→  start: number,
   138→  duration: number,
   139→  volume = 0.09,
   140→  type: OscillatorType = "sine",
   141→  bus: GainNode | null = null,
   142→  glideTo?: number,
   143→) {
   144→  const ac = ensureCtx();
   145→  if (!ac || !enabled) return;
   146→  const osc = ac.createOscillator();
   147→  const gain = ac.createGain();
   148→  osc.type = type;
   149→  const t = ac.currentTime + start;
   150→  osc.frequency.setValueAtTime(freq, t);
   151→  if (glideTo) osc.frequency.exponentialRampToValueAtTime(Math.max(20, glideTo), t + duration);
   152→  gain.gain.setValueAtTime(0.0001, t);
   153→  gain.gain.exponentialRampToValueAtTime(Math.max(volume, 0.0002), t + 0.012);
   154→  gain.gain.exponentialRampToValueAtTime(0.0001, t + duration);
   155→  osc.connect(gain).connect(bus ?? sfxBus ?? ac.destination);
   156→  osc.start(t);
   157→  osc.stop(t + duration + 0.06);
   158→}
   159→
   160→/** Campanada tipo campana con parciales inarmónicos (para bells y éxito). */
   161→function bell(freq: number, start: number, duration = 0.9, volume = 0.08) {
   162→  const ac = ensureCtx();
   163→  if (!ac || !enabled) return;
   164→  const partials = [1, 2.76, 5.4];
   165→  partials.forEach((p, i) => {
   166→    tone(freq * p, start, duration / (1 + i * 0.4), volume / (i + 1.5), "sine");
   167→  });
   168→}

   170→/** Ráfaga de ruido filtrada (clics, olas, viento, aplausos). */
   171→function noiseBurst(
   172→  start: number,
   173→  duration: number,
   174→  filterFreq: number,
   175→  volume = 0.06,
   176→  type: BiquadFilterType = "bandpass",
   177→  q = 1,
   178→  bus: GainNode | null = null,
   179→) {
   180→  const ac = ensureCtx();
   181→  if (!ac || !enabled || !noiseBuffer) return;
   182→  const src = ac.createBufferSource();
   183→  src.buffer = noiseBuffer;
   184→  src.loop = true;
   185→  const filter = ac.createBiquadFilter();
   186→  filter.type = type;
   187→  filter.frequency.value = filterFreq;
   188→  filter.Q.value = q;
   189→  const gain = ac.createGain();
   190→  const t = ac.currentTime + start;
   191→  gain.gain.setValueAtTime(0.0001, t);
   192→  gain.gain.exponentialRampToValueAtTime(volume, t + Math.min(0.05, duration / 3));
   193→  gain.gain.exponentialRampToValueAtTime(0.0001, t + duration);
   194→  src.connect(filter).connect(gain).connect(bus ?? sfxBus ?? ac.destination);
   195→  src.start(t, Math.random() * 1.5);
   196→  src.stop(t + duration + 0.08);
   197→}

   374→const semitone = (root: number, s: number) => root * Math.pow(2, s / 12);

   376→/** Funda de entrada a la escena: acorde inicial y textura. */
   377→function startScene(sceneId: string) {
   378→  const ac = ensureCtx();
   379→  if (!ac || !enabled) return;
   380→  const scene = SCENES[sceneId] ?? SCENES.home;
   381→  currentScene = sceneId;
   382→  stopTexture();
   383→  startTexture(scene);
   384→  // Acorde de bienvenida
   385→  scene.chords[0].forEach((d) => padNote(semitone(scene.root, scene.scale[d]), 0.02, 6, 0.05));
   386→  if (!schedulerId) {
   387→    nextBeatTime = ac.currentTime + 0.1;
   388→    beatIndex = 0;
   389→    barIndex = 0;
   390→    schedulerId = setInterval(scheduleAhead, 240);
   391→  }
   392→}

   394→/** Cambia la música de escena con fundido. La escena ya activa no se re-inicia. */
   395→export function setSceneMusic(sceneId: string | null) {
   396→  if (!sceneId || sceneId === currentScene) return;
   397→  if (!ctx || !unlocked) {
   398→    currentScene = sceneId; // se aplicará al desbloquear
   399→    return;
   400→  }
   401→  startScene(sceneId);
   402→}
   403→
   404→function stopTexture() {
   405→  if (textureNodes && ctx) {
   406→    try {
   407→      textureNodes.gain.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.4);
   408→      const src = textureNodes.src;
   409→      setTimeout(() => {
   410→        try {
   411→          src.stop();
   412→        } catch {
   413→          /* ya parado */
   414→        }
   415→      }, 1500);
   416→    } catch {
   417→      /* ignorar */
   418→    }
   419→    textureNodes = null;
   420→  }
   421→}

   423→function startTexture(scene: SceneMusic) {
   424→  const ac = ctx;
   425→  if (!ac || !noiseBuffer) return;
   426→  const src = ac.createBufferSource();
   427→  src.buffer = noiseBuffer;
   428→  src.loop = true;
   429→  const filter = ac.createBiquadFilter();
   430→  const gain = ac.createGain();
   431→  gain.gain.value = 0.0001;
   432→  const t = ac.currentTime;
   433→  let lfo: OscillatorNode | null = null;
   434→  let lfoGain: GainNode | null = null;
   435→
   436→  switch (scene.texture) {
   437→    case "wind":
   438→      filter.type = "lowpass";
   439→      filter.frequency.value = 420;
   440→      // ráfagas lentas
   441→      lfo = ac.createOscillator();
   442→      lfo.frequency.value = 0.09;
   443→      lfoGain = ac.createGain();
   444→      lfoGain.gain.value = scene.textureLevel * 0.7;
   445→      lfo.connect(lfoGain).connect(gain.gain);
   446→      gain.gain.value = scene.textureLevel;
   447→      break;
   448→    case "waves":
   449→      filter.type = "lowpass";
   450→      filter.frequency.value = 300;
   451→      lfo = ac.createOscillator();
   452→      lfo.frequency.value = 0.07; // marea muy lenta
   453→      lfoGain = ac.createGain();
   454→      lfoGain.gain.value = scene.textureLevel * 0.8;
   455→      lfo.connect(lfoGain).connect(gain.gain);
   456→      gain.gain.value = scene.textureLevel;
   457→      break;
   458→    case "hum":
   459→      filter.type = "lowpass";
   460→      filter.frequency.value = 180;
   461→      gain.gain.setTargetAtTime(scene.textureLevel, t, 1.2);
   462→      break;
   463→    case "paper":
   464→      filter.type = "bandpass";
   465→      filter.frequency.value = 4200;
   466→      filter.Q.value = 0.4;
   467→      gain.gain.setTargetAtTime(scene.textureLevel * 0.5, t, 1.5);
   468→      break;
   469→    case "crackle":
   470→      filter.type = "bandpass";
   471→      filter.frequency.value = 2600;
   472→      filter.Q.value = 2.5;
   473→      lfo = ac.createOscillator();
   474→      lfo.frequency.value = 1.7;
   475→      lfoGain = ac.createGain();
   476→      lfoGain.gain.value = scene.textureLevel;
   477→      lfo.connect(lfoGain).connect(gain.gain);
   478→      gain.gain.value = scene.textureLevel * 0.3;
   479→      break;
   480→    case "air":
   481→    default:
   482→      filter.type = "lowpass";
   483→      filter.frequency.value = 900;
   484→      gain.gain.setTargetAtTime(scene.textureLevel, t, 1.5);
   485→      break;
   486→  }
   487→  src.connect(filter).connect(gain).connect(noiseBus ?? master ?? ac.destination);
   488→  if (lfo && lfoGain) lfo.start();
   489→  src.start();
   490→  textureNodes = { src, gain };
   491→}

   493→/** Colchón armónico suave (pad) con ataque largo y filtro móvil. */
   494→function padNote(freq: number, start: number, duration: number, volume: number) {
   495→  const ac = ctx;
   496→  if (!ac || !enabled) return;
   497→  const t = ac.currentTime + start;
   498→  const gain = ac.createGain();
   499→  gain.gain.setValueAtTime(0.0001, t);
   500→  gain.gain.linearRampToValueAtTime(volume, t + duration * 0.35);
   501→  gain.gain.setTargetAtTime(0.0001, t + duration * 0.6, 1.2);
   502→  const filter = ac.createBiquadFilter();
   503→  filter.type = "lowpass";
   504→  filter.frequency.setValueAtTime(freq * 1.4, t);
   505→  filter.frequency.linearRampToValueAtTime(freq * 0.9, t + duration);
   506→  [0, 4, -5].forEach((detune) => {
   507→    const osc = ac.createOscillator();
   508→    osc.type = "sawtooth";
   509→    osc.frequency.value = freq;
   510→    osc.detune.value = detune * 3;
   511→    osc.connect(filter);
   512→    osc.start(t);
   513→    osc.stop(t + duration + 1.8);
   514→  });
   515→  filter.connect(gain).connect(musicBus ?? ac.destination);
   516→}

   518→/** Nota melódica breve (arpegio/pltuck) con carácter de la escena. */
   519→function pluck(freq: number, start: number, volume: number, type: OscillatorType) {
   520→  const ac = ctx;
   521→  if (!ac || !enabled) return;
   522→  const t = ac.currentTime + start;
   523→  const osc = ac.createOscillator();
   524→  const gain = ac.createGain();
   525→  osc.type = type;
   526→  osc.frequency.value = freq;
   527→  gain.gain.setValueAtTime(0.0001, t);
   528→  gain.gain.exponentialRampToValueAtTime(volume, t + 0.02);
   529→  gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.55);
   530→  osc.connect(gain).connect(musicBus ?? ac.destination);
   531→  osc.start(t);
   532→  osc.stop(t + 0.65);
   533→}

   535→/** El tic del Gran Reloj: madera seca y corta. */
   536→function clockTick(high: boolean, start: number, volume: number) {
   537→  const ac = ctx;
   538→  if (!ac || !enabled) return;
   539→  const t = ac.currentTime + start;
   540→  const osc = ac.createOscillator();
   541→  const gain = ac.createGain();
   542→  osc.type = "square";
   543→  osc.frequency.setValueAtTime(high ? 1980 : 1560, t);
   544→  osc.frequency.exponentialRampToValueAtTime(high ? 1500 : 1180, t + 0.03);
   545→  gain.gain.setValueAtTime(0.0001, t);
   546→  gain.gain.exponentialRampToValueAtTime(volume, t + 0.004);
   547→  gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.05);
   548→  const filter = ac.createBiquadFilter();
   549→  filter.type = "bandpass";
   550→  filter.frequency.value = 1900;
   551→  filter.Q.value = 4;
   552→  osc.connect(filter).connect(gain).connect(clockBus ?? ac.destination);
   553→  osc.start(t);
   554→  osc.stop(t + 0.07);
   555→  // chasquido de mecánica
   556→  noiseBurst(start, 0.018, 3400, volume * 0.4, "bandpass", 5, clockBus);
   557→}

   559→function scheduleAhead() {
   560→  const ac = ctx;
   561→  const scene = SCENES[currentScene ?? "home"] ?? SCENES.home;
   562→  if (!ac || !enabled) return;
   563→  const beatDur = 60 / scene.bpm;
   564→  while (nextBeatTime < ac.currentTime + 0.6) {
   565→    const start = Math.max(0, nextBeatTime - ac.currentTime);
   566→    const beatInBar = beatIndex % 4;
   567→
   568→    /* -- pulso: colchón en cada compás, bajo en el 1 -- */
   569→    if (beatInBar === 0) {
   570→      const chord = scene.chords[barIndex % scene.chords.length];
   571→      chord.forEach((d) =>
   572→        padNote(semitone(scene.root * 2, scene.scale[d]), start, beatDur * 4.2, 0.035),
   573→      );
   574→      if (scene.bass) padNote(scene.root / 2, start, beatDur * 4, 0.05);
   575→    }
   576→
   577→    /* -- arpegios en los contratiempos (nunca en el pulso fuerte) -- */
   578→    const offbeat = start + beatDur * 0.5;
   579→    if (Math.random() < scene.pluck) {
   580→      const deg = scene.scale[Math.floor(Math.random() * scene.scale.length)];
   581→      const octave = Math.random() < 0.3 ? 24 : 12;
   582→      pluck(semitone(scene.root, deg + octave), offbeat, 0.055, scene.pluckType);
   583→    }
   584→    if (beatInBar === 2 && Math.random() < scene.pluck * 0.7) {
   585→      const deg = scene.scale[Math.floor(Math.random() * scene.scale.length)];
   586→      pluck(semitone(scene.root, deg + 12), start + beatDur * 0.25, 0.04, "sine");
   587→    }
   588→
   589→    /* -- EL GRAN RELOJ EN CONTRATIEMPO --
   590→       tic-tac entre pulsos: en reposo muy sutil; con tensión se
   591→       vuelve insistente (cada contratiempo) y sube de tono. */
   592→    const clockBase = clockOn ? 0.055 : 0.018;
   593→    const clockVol = clockBase * (0.35 + tension * 1.65) * (beatInBar % 2 === 0 ? 1 : 0.8);
   594→    clockTick(clockTickToggle % 2 === 0, offbeat, clockVol);
   595→    clockTickToggle++;
   596→    if (tension > 0.55) {
   597→      // doble tic nervioso en el segundo contratiempo del pulso
   598→      clockTick(clockTickToggle % 2 === 0, offbeat + beatDur * 0.25, clockVol * 0.6);
   599→    }
   600→
   601→    beatIndex++;
   602→    if (beatInBar === 3) barIndex++;
   603→    nextBeatTime += beatDur;
   604→  }
   605→}

   608→export function setClockActive(active: boolean) {
   609→  clockOn = active;

   612→/** Tensión 0-1: acelera el reloj y lo vuelve protagonista. */
   613→export function setTension(level: number) {
   614→  tension = Math.min(1, Math.max(0, level));
   615→  if (clockBus && ctx) {
   616→    clockBus.gain.setTargetAtTime(0.35 + tension * 0.65, ctx.currentTime, 0.12);
   617→  }
   618→}

   820→/** ¿Soporta este navegador síntesis de voz en español? */
   821→export function ttsAvailable(): boolean {
   822→  if (typeof window === "undefined" || !("speechSynthesis" in window)) return false;
   823→  return true;
   824→}
   825→
   826→/**
   827→ * Lee un texto en voz alta (es-ES) si el navegador lo permite.
   828→ * Devuelve false si no está disponible: la UI debe ofrecer siempre
   829→ * la alternativa visual.
   830→ */
   831→export function speak(text: string, rate = 0.95): boolean {
   832→  if (!ttsAvailable()) return false;
   833→  try {
   834→    window.speechSynthesis.cancel();
   835→    const utter = new SpeechSynthesisUtterance(text);
   836→    utter.lang = "es-ES";
   837→    utter.rate = rate;
   838→    window.speechSynthesis.speak(utter);
   839→    return true;
   840→  } catch {
   841→    return false;
   842→  }
   843→}
