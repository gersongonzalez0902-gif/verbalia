   201→/* ============================== Contenido ============================== */

   203→interface ExerciseBase {
   204→  /** Identificador único y estable del ejercicio. */
   205→  id: string;
   206→  /** Habilidad principal que entrena. */
   207→  skill: SkillId;
   208→  /** Dificultad 1-5. */
   209→  difficulty: Difficulty;
   210→  /** Referencia MCER orientativa. */
   211→  cefr: Cefr;
   212→  /** Consigna específica del ejercicio. */
   213→  prompt: string;
   214→  /** Explicación breve de la regla o el porqué (retroalimentación). */
   215→  explanation: string;
   216→  /** Etiquetas temáticas. */
   217→  tags: string[];
   218→}
