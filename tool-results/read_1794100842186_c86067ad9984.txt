     1→/**
     2→ * Motor de audio de VERBALIA · v2
     3→ * ---------------------------------------------------------------
     4→ * Todo el sonido del juego se sintetiza aquí con WebAudio:
     5→ * sin ficheros, sin descargas, sin red.
     6→ *
     7→ * 1) SFX: ~30 efectos de interacción (acierto, fallo, niveles,
     8→ *    monedas de XP, naipes, etc.) con variantes para que nunca
     9→ *    suenen igual dos veces seguidas.
    10→ * 2) MÚSICA AMBIENTE GENERATIVA: cada escena/ región tiene su
    11→ *    propia paleta musical (escala, tempo, timbres, textura de
    12→ *    ruido) compuesta en tiempo real con un secuenciador por
    13→ *    adelantado (lookahead). Al cambiar de región la música
    14→ *    hace fundido cruzado.
    15→ * 3) EL GRAN RELOJ EN CONTRATIEMPO: una capa de tic-tac que
    16→ *    suena DESFASADA respecto al pulso musical (en los
    17→ *    contratiempos, como pidió el jugador). En reposo es una
    18→ *    presencia mínima y sugerente; en los juegos cronometrados
    19→ *    `setTension()` la enciende, la acelera y le sube el tono
    20→ *    hasta el clímax de los últimos segundos.
    21→ *
    22→ * El AudioContext se desbloquea con el primer gesto del usuario
    23→ * (política de navegadores). Si el jugador silencia el juego,
    24→ * el contexto se suspende y todo queda en silencio absoluto.
    25→ */
    26→
    27→/* ============================== ajustes ============================== */
    28→
    29→let ctx: AudioContext | null = null;
    30→let master: GainNode | null = null;
    31→let musicBus: GainNode | null = null;
    32→let sfxBus: GainNode | null = null;
    33→let clockBus: GainNode | null = null;
    34→let noiseBus: GainNode | null = null;
    35→
    36→let enabled = true;
    37→let musicVolume = 0.55;
    38→let sfxVolume = 0.9;
    39→
    40→let unlocked = false;
    41→let unlockListeners: (() => void) | null = null;
    42→
    43→/* Ruido compartido (buffer de 2 s reutilizado por todas las texturas) */
    44→let noiseBuffer: AudioBuffer | null = null;
    45→
    46→function ensureCtx(): AudioContext | null {
    47→  if (typeof window === "undefined") return null;
    48→  try {
    49→    if (!ctx) {
    50→      const AudioCtor =
    51→        window.AudioContext ??
    52→        (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
    53→      if (!AudioCtor) return null;
    54→      ctx = new AudioCtor();
    55→      master = ctx.createGain();
    56→      master.gain.value = enabled ? 0.9 : 0;
    57→      master.connect(ctx.destination);
    58→
    59→      musicBus = ctx.createGain();
    60→      musicBus.gain.value = musicVolume;
    61→      musicBus.connect(master);
    62→
    63→      sfxBus = ctx.createGain();
    64→      sfxBus.gain.value = sfxVolume;
    65→      sfxBus.connect(master);
    66→
    67→      clockBus = ctx.createGain();
    68→      clockBus.gain.value = 0.5;
    69→      clockBus.connect(master);
    70→
    71→      noiseBus = ctx.createGain();
    72→      noiseBus.gain.value = 1;
    73→      noiseBus.connect(musicBus);
    74→
    75→      const len = ctx.sampleRate * 2;
    76→      noiseBuffer = ctx.createBuffer(1, len, ctx.sampleRate);
    77→      const data = noiseBuffer.getChannelData(0);
    78→      for (let i = 0; i < len; i++) data[i] = Math.random() * 2 - 1;
    79→    }
    80→    if (ctx.state === "suspended" && enabled) void ctx.resume();
    81→    return ctx;
    82→  } catch {
    83→    return null;
    84→  }
    85→}
    86→
    87→/** Desbloquea el audio en el primer gesto (tap, teclado, scroll táctil). */
    88→export function initAudioUnlock(): void {
    89→  if (typeof window === "undefined" || unlockListeners) return;
    90→  const unlock = () => {
    91→    ensureCtx();
    92→    unlocked = true;
    93→    if (currentScene) startScene(currentScene); // la música arranca al desbloquear
    94→    detach();
    95→  };
    96→  const detach = () => {
    97→    window.removeEventListener("pointerdown", unlock);
    98→    window.removeEventListener("keydown", unlock);
    99→    window.removeEventListener("touchstart", unlock);
   100→    unlockListeners = null;
   101→  };
   102→  window.addEventListener("pointerdown", unlock, { once: false, passive: true });
   103→  window.addEventListener("keydown", unlock, { once: false });
   104→  window.addEventListener("touchstart", unlock, { once: false, passive: true });
   105→  unlockListeners = detach;
   106→  registerVisibilityGuard();
   107→}

   109→/* ================== guarda de visibilidad de la página ==================
   110→ * Cuando la pestaña se oculta o el teléfono se suspende, la música
   111→ * generativa debe callar (y el AudioContext congelarse para no gastar
   112→ * batería). Al volver, la escena se reanuda si sonaba antes. */
   113→
   114→let visibilityGuardOn = false;
   115→let pausedByVisibility = false;
   116→
   117→export function registerVisibilityGuard(): void {
   118→  if (typeof window === "undefined" || visibilityGuardOn) return;
   119→  visibilityGuardOn = true;
   120→
   121→  const pauseAll = () => {
   122→    if (pausedByVisibility) return;
   123→    pausedByVisibility = true;
   124→    // 1) Banda sonora generativa: se detiene el secuenciador y la textura
   125→    //    (la escena se recuerda para reanudarla al volver).
   126→    if (schedulerId) {
   127→      clearInterval(schedulerId);
   128→      schedulerId = null;
   129→    }
   130→    stopTexture();
   131→    // 2) El contexto entero se suspende: cero audio y cero batería mientras
   132→    //    la página no está visible (móvil bloqueado, app en segundo plano…).
   133→    if (ctx && ctx.state === "running") void ctx.suspend();
   134→  };
   135→
   136→  const resumeAll = () => {
   137→    if (!pausedByVisibility) return;
   138→    pausedByVisibility = false;
   139→    if (!enabled) return; // el jugador silenció el juego mientras no miraba
   140→    if (ctx && ctx.state === "suspended") void ctx.resume();
   141→    // La escena que sonaba antes del ocultado vuelve con fundido.
   142→    if (currentScene && unlocked) startScene(currentScene);
   143→  };
   144→
   145→  document.addEventListener("visibilitychange", () => {
   146→    if (document.visibilityState === "hidden") pauseAll();
   147→    else resumeAll();
   148→  });
   149→  // iOS y algunos navegadores: al minimizar o navegar fuera de la página.
   150→  window.addEventListener("pagehide", () => pauseAll());
   151→  window.addEventListener("pageshow", () => resumeAll());
   152→  // Congelar la pestaña (procesos en background) también detiene el audio.
   153→  document.addEventListener("freeze" as keyof DocumentEventMap, () => pauseAll());
   154→  document.addEventListener("resume" as keyof DocumentEventMap, () => resumeAll());
   155→}
   156→
   157→export function setSoundEnabled(value: boolean) {
   158→  enabled = value;
   159→  const ac = ctx ?? null;
   160→  if (ac && master) master.gain.setTargetAtTime(value ? 0.9 : 0, ac.currentTime, 0.05);
   161→  if (!value && ac && ac.state === "running") void ac.suspend();
   162→  if (value && ac && ac.state === "suspended") {
   163→    void ac.resume();
   164→    unlocked = true;
   165→  }
   166→}
   167→
   168→export function setMusicVolume(v: number) {
   169→  musicVolume = Math.min(1, Math.max(0, v));
   170→  if (ctx && musicBus) musicBus.gain.setTargetAtTime(musicVolume, ctx.currentTime, 0.08);
   171→}
   172→
   173→export function setSfxVolume(v: number) {
   174→  sfxVolume = Math.min(1, Math.max(0, v));
   175→  if (ctx && sfxBus) sfxBus.gain.setTargetAtTime(sfxVolume, ctx.currentTime, 0.05);
   176→}
   177→
   178→export function isAudioUnlocked(): boolean {
   179→  return unlocked && !!ctx && ctx.state === "running";
   180→}

   182→/* ============================== voces base ============================== */
   183→
   184→function tone(
   185→  freq: number,
   186→  start: number,
   187→  duration: number,
   188→  volume = 0.09,
   189→  type: OscillatorType = "sine",
   190→  bus: GainNode | null = null,
   191→  glideTo?: number,
   192→) {
   193→  const ac = ensureCtx();
   194→  if (!ac || !enabled) return;
   195→  const osc = ac.createOscillator();
   196→  const gain = ac.createGain();
   197→  osc.type = type;
   198→  const t = ac.currentTime + start;
   199→  osc.frequency.setValueAtTime(freq, t);
   200→  if (glideTo) osc.frequency.exponentialRampToValueAtTime(Math.max(20, glideTo), t + duration);
   201→  gain.gain.setValueAtTime(0.0001, t);
   202→  gain.gain.exponentialRampToValueAtTime(Math.max(volume, 0.0002), t + 0.012);
   203→  gain.gain.exponentialRampToValueAtTime(0.0001, t + duration);
   204→  osc.connect(gain).connect(bus ?? sfxBus ?? ac.destination);
   205→  osc.start(t);
   206→  osc.stop(t + duration + 0.06);
   207→}
   208→
   209→/** Campanada tipo campana con parciales inarmónicos (para bells y éxito). */
   210→function bell(freq: number, start: number, duration = 0.9, volume = 0.08) {
   211→  const ac = ensureCtx();
   212→  if (!ac || !enabled) return;
   213→  const partials = [1, 2.76, 5.4];
   214→  partials.forEach((p, i) => {
   215→    tone(freq * p, start, duration / (1 + i * 0.4), volume / (i + 1.5), "sine");
   216→  });
   217→}
   218→
   219→/** Ráfaga de ruido filtrada (clics, olas, viento, aplausos). */
   220→function noiseBurst(
   221→  start: number,
   222→  duration: number,
   223→  filterFreq: number,
   224→  volume = 0.06,
   225→  type: BiquadFilterType = "bandpass",
   226→  q = 1,
   227→  bus: GainNode | null = null,
   228→) {
   229→  const ac = ensureCtx();
   230→  if (!ac || !enabled || !noiseBuffer) return;
   231→  const src = ac.createBufferSource();
   232→  src.buffer = noiseBuffer;
   233→  src.loop = true;
   234→  const filter = ac.createBiquadFilter();
   235→  filter.type = type;
   236→  filter.frequency.value = filterFreq;
   237→  filter.Q.value = q;
   238→  const gain = ac.createGain();
   239→  const t = ac.currentTime + start;
   240→  gain.gain.setValueAtTime(0.0001, t);
   241→  gain.gain.exponentialRampToValueAtTime(volume, t + Math.min(0.05, duration / 3));
   242→  gain.gain.exponentialRampToValueAtTime(0.0001, t + duration);
   243→  src.connect(filter).connect(gain).connect(bus ?? sfxBus ?? ac.destination);
   244→  src.start(t, Math.random() * 1.5);
   245→  src.stop(t + duration + 0.08);
   246→}

   646→/** Funda de entrada a la escena: acorde inicial y textura. */
   647→function startScene(sceneId: string) {
   648→  const ac = ensureCtx();
   649→  if (!ac || !enabled) return;
   650→  const scene = SCENES[sceneId] ?? SCENES.home;
   651→  currentScene = sceneId;
   652→  // Si la escena anterior se silenció (stopSceneMusic), recupera el volumen.
   653→  if (musicBus) musicBus.gain.setTargetAtTime(musicVolume, ac.currentTime, 0.5);
   654→  stopTexture();
   655→  startTexture(scene);
   656→  // Acorde de bienvenida
   657→  scene.chords[0].forEach((d) => padNote(semitone(scene.root, scene.scale[d]), 0.02, 6, 0.05));
   658→  if (!schedulerId) {
   659→    nextBeatTime = ac.currentTime + 0.1;
   660→    beatIndex = 0;
   661→    barIndex = 0;
   662→    schedulerId = setInterval(scheduleAhead, 240);
   663→  }
   664→}

   668→  if (!sceneId || sceneId === currentScene) return;

   670→    currentScene = sceneId; // se aplicará al desbloquear
   671→    return;

   688→function stopTexture() {
   689→  if (textureNodes && ctx) {
   690→    try {
   691→      textureNodes.gain.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.4);
   692→      const src = textureNodes.src;
   693→      setTimeout(() => {
   694→        try {
   695→          src.stop();
   696→        } catch {
   697→          /* ya parado */
   698→        }
   699→      }, 1500);
   700→    } catch {
   701→      /* ignorar */
   702→    }
   703→    textureNodes = null;
   704→  }
   705→}

   707→function startTexture(scene: SceneMusic) {
   708→  // 🎵 Retirado por petición del navegante: sin ruido de lluvia, viento ni
   709→  // ambiente de fondo. La banda sonora queda solo con la parte melódica
   710→  // (pads, arpegios y bajo), mucho más limpia y agradable.
   711→  void scene;
   712→  void noiseBus;
   713→  return;
   714→}

   716→/** Colchón armónico suave (pad) con ataque largo y filtro móvil. */
   717→function padNote(freq: number, start: number, duration: number, volume: number) {
   718→  const ac = ctx;
   719→  if (!ac || !enabled) return;
   720→  const t = ac.currentTime + start;
   721→  const gain = ac.createGain();
   722→  gain.gain.setValueAtTime(0.0001, t);
   723→  gain.gain.linearRampToValueAtTime(volume, t + duration * 0.35);
   724→  gain.gain.setTargetAtTime(0.0001, t + duration * 0.6, 1.2);
   725→  const filter = ac.createBiquadFilter();
   726→  filter.type = "lowpass";
   727→  filter.frequency.setValueAtTime(freq * 1.4, t);
   728→  filter.frequency.linearRampToValueAtTime(freq * 0.9, t + duration);
   729→  [0, 4, -5].forEach((detune) => {
   730→    const osc = ac.createOscillator();
   731→    osc.type = "sawtooth";
   732→    osc.frequency.value = freq;
   733→    osc.detune.value = detune * 3;
   734→    osc.connect(filter);
   735→    osc.start(t);
   736→    osc.stop(t + duration + 1.8);
   737→  });
   738→  filter.connect(gain).connect(musicBus ?? ac.destination);
   739→}

   741→/** Nota melódica breve (arpegio/pltuck) con carácter de la escena. */
   742→function pluck(freq: number, start: number, volume: number, type: OscillatorType) {
   743→  const ac = ctx;
   744→  if (!ac || !enabled) return;
   745→  const t = ac.currentTime + start;
   746→  const osc = ac.createOscillator();
   747→  const gain = ac.createGain();
   748→  osc.type = type;
   749→  osc.frequency.value = freq;
   750→  gain.gain.setValueAtTime(0.0001, t);
   751→  gain.gain.exponentialRampToValueAtTime(volume, t + 0.02);
   752→  gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.55);
   753→  osc.connect(gain).connect(musicBus ?? ac.destination);
   754→  osc.start(t);
   755→  osc.stop(t + 0.65);
   756→}

   758→/** El tic del Gran Reloj: madera seca y corta. */
   759→function clockTick(high: boolean, start: number, volume: number) {
   760→  const ac = ctx;
   761→  if (!ac || !enabled) return;
   762→  const t = ac.currentTime + start;
   763→  const osc = ac.createOscillator();
   764→  const gain = ac.createGain();
   765→  osc.type = "square";
   766→  osc.frequency.setValueAtTime(high ? 1980 : 1560, t);
   767→  osc.frequency.exponentialRampToValueAtTime(high ? 1500 : 1180, t + 0.03);
   768→  gain.gain.setValueAtTime(0.0001, t);
   769→  gain.gain.exponentialRampToValueAtTime(volume, t + 0.004);
   770→  gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.05);
   771→  const filter = ac.createBiquadFilter();
   772→  filter.type = "bandpass";
   773→  filter.frequency.value = 1900;
   774→  filter.Q.value = 4;
   775→  osc.connect(filter).connect(gain).connect(clockBus ?? ac.destination);
   776→  osc.start(t);
   777→  osc.stop(t + 0.07);
   778→  // chasquido de mecánica
   779→  noiseBurst(start, 0.018, 3400, volume * 0.4, "bandpass", 5, clockBus);
   780→}

   782→function scheduleAhead() {
   783→  const ac = ctx;
   784→  const scene = SCENES[currentScene ?? "home"] ?? SCENES.home;
   785→  if (!ac || !enabled) return;
   786→  const beatDur = 60 / scene.bpm;
   787→  while (nextBeatTime < ac.currentTime + 0.6) {
   788→    const start = Math.max(0, nextBeatTime - ac.currentTime);
   789→    const beatInBar = beatIndex % 4;
   790→
   791→    /* -- pulso: colchón en cada compás, bajo en el 1 -- */
   792→    if (beatInBar === 0) {
   793→      const chord = scene.chords[barIndex % scene.chords.length];
   794→      chord.forEach((d) =>
   795→        padNote(semitone(scene.root * 2, scene.scale[d % scene.scale.length]), start, beatDur * 4.2, 0.035),
   796→      );
   797→      if (scene.bass) padNote(scene.root / 2, start, beatDur * 4, 0.05);
   798→    }
   799→
   800→    /* -- arpegios en los contratiempos (nunca en el pulso fuerte) -- */
   801→    const offbeat = start + beatDur * 0.5;
   802→    if (Math.random() < scene.pluck) {
   803→      const deg = scene.scale[Math.floor(Math.random() * scene.scale.length)];
   804→      const octave = Math.random() < 0.3 ? 24 : 12;
   805→      pluck(semitone(scene.root, deg + octave), offbeat, 0.055, scene.pluckType);
   806→    }
   807→    if (beatInBar === 2 && Math.random() < scene.pluck * 0.7) {
   808→      const deg = scene.scale[Math.floor(Math.random() * scene.scale.length)];
   809→      pluck(semitone(scene.root, deg + 12), start + beatDur * 0.25, 0.04, "sine");
   810→    }
   811→
   812→    /* -- EL GRAN RELOJ EN CONTRATIEMPO --
   813→       tic-tac entre pulsos: en reposo muy sutil; con tensión se
   814→       vuelve insistente (cada contratiempo) y sube de tono. */
   815→    const clockBase = clockOn ? 0.055 : 0.018;
   816→    const clockVol = clockBase * (0.35 + tension * 1.65) * (beatInBar % 2 === 0 ? 1 : 0.8);
   817→    clockTick(clockTickToggle % 2 === 0, offbeat, clockVol);
   818→    clockTickToggle++;
   819→    if (tension > 0.55) {
   820→      // doble tic nervioso en el segundo contratiempo del pulso
   821→      clockTick(clockTickToggle % 2 === 0, offbeat + beatDur * 0.25, clockVol * 0.6);
   822→    }
   823→
   824→    beatIndex++;
   825→    if (beatInBar === 3) barIndex++;
   826→    nextBeatTime += beatDur;
   827→  }
   828→}

  1094→export function ttsAvailable(): boolean {

  1102→/** Fija la velocidad global del TTS (0.5 - 1.3). */

  1114→/** Vibra un patrón corto (aciertos y combos). Silencioso si no hay soporte. */
  1115→export function haptic(pattern: number | number[] = 12) {

  1117→  try {
  1118→    if (typeof navigator !== "undefined" && "vibrate" in navigator) {
  1119→      navigator.vibrate(pattern);
  1120→    }
  1121→  } catch {
  1122→    /* sin soporte: silencio */
  1123→  }
  1124→}

  1131→/**
  1132→ * Lee un texto en voz alta (es-ES) si el navegador lo permite.
  1133→ * Devuelve false si no está disponible: la UI debe ofrecer siempre
  1134→ * la alternativa visual.
  1135→ */
  1136→export function speak(text: string, rate = 0.95): boolean {
  1137→  if (!ttsAvailable()) return false;
  1138→  try {
  1139→    window.speechSynthesis.cancel();
  1140→    const utter = new SpeechSynthesisUtterance(text);
  1141→    utter.lang = "es-ES";
  1142→    utter.rate = rate * ttsRateGlobal;
  1143→    window.speechSynthesis.speak(utter);
  1144→    return true;
  1145→  } catch {
  1146→    return false;
  1147→  }
  1148→}
