   631→/* ============================ Bóveda cifrada ============================ */
   632→
   633→function VaultSection() {
   634→  const exportJson = useGameStore((s) => s.exportJson);
   635→  const importJson = useGameStore((s) => s.importJson);
   636→  const { toast } = useToast();
   637→  const [pass, setPass] = useState("");
   638→  const [code, setCode] = useState("");
   639→  const [busy, setBusy] = useState(false);
   640→  const [lastCode, setLastCode] = useState<string | null>(null);
   641→
   642→  const saveToVault = async () => {
   643→    if (pass.length < 8) {
   644→      toast({ title: "Contraseña corta", description: "Usa al menos 8 caracteres: cierra bien el sobre.", variant: "destructive" });
   645→      return;
   646→    }
   647→    const json = exportJson();
   648→    if (!json) return;
   649→    setBusy(true);
   650→    try {
   651→      const { id, expiresAt } = await uploadVault(json, pass);
   652→      setLastCode(id);
   653→      toast({
   654→        title: "Sobre guardado en la bóveda",
   655→        description: `Código ${id.slice(0, 10)}… Caduca el ${new Date(expiresAt).toLocaleDateString("es-ES")}. Anótalo junto a tu contraseña.`,
   656→      });
   657→    } catch (e) {
   658→      toast({ title: "No se pudo guardar", description: e instanceof Error ? e.message : "Error inesperado", variant: "destructive" });
   659→    } finally {
   660→      setBusy(false);
   661→    }
   662→  };
   663→
   664→  const restoreFromVault = async () => {
   665→    if (!/^\\S+$/.test(code.trim())) {
   666→      toast({ title: "Falta el código", description: "Pega el código de 64 caracteres que te dieron al guardar.", variant: "destructive" });
   667→      return;
   668→    }
   669→    if (!pass) {
   670→      toast({ title: "Falta la contraseña", description: "La misma con la que sellaste el sobre.", variant: "destructive" });
   671→      return;
   672→    }
   673→    setBusy(true);
   674→    try {
   675→      const json = await downloadVault(code.trim(), pass);
   676→      const ok = await importJson(json);
   677→      toast({
   678→        title: ok ? "¡Bóveda abierta!" : "Sobre corrupto",
   679→        description: ok ? "Tu progreso ha regresado del viaje." : "El contenido no era una copia de VerbalIA.",
   680→        variant: ok ? undefined : "destructive",
   681→      });
   682→    } catch (e) {
   683→      toast({ title: "La bóveda no abrió", description: e instanceof Error ? e.message : "Error inesperado", variant: "destructive" });
   684→    } finally {
   685→      setBusy(false);
   686→    }
   687→  };
   688→
   689→  return (
   690→    <section className="card-aurora rounded-3xl border bg-card p-5" aria-labelledby="boveda">
   691→      <h2 id="boveda" className="font-display flex items-center gap-2 text-lg font-bold">
   692→        <Lock className="h-5 w-5 text-primary" aria-hidden /> Bóveda cifrada (zero-knowledge)
   693→      </h2>
   694→      <p className="mt-1 text-xs leading-relaxed text-muted-foreground">
   695→        Tu progreso se cifra <strong>en este dispositivo</strong> (AES-GCM, PBKDF2 con 600 000
   696→        iteraciones) y el servidor solo guarda un sobre sellado que <em>no puede</em> abrir:
   697→        ni la contraseña ni el contenido viajan en claro. Ideal como copia de emergencia
   698→        con una contraseña que recuerdes.
   699→      </p>
   700→
   701→      <div className="mt-4 space-y-3">
   702→        <div>
   703→          <Label htmlFor="vault-pass" className="text-sm font-bold">Contraseña del sobre</Label>
   704→          <input
   705→            id="vault-pass"
   706→            type="password"
   707→            value={pass}
   708→            onChange={(e) => setPass(e.target.value)}
   709→            autoComplete="new-password"
   710→            aria-label="Contraseña de la bóveda"
   711→            className="mt-1.5 h-10 w-full rounded-2xl border border-border bg-card px-3 text-sm font-semibold outline-none focus:border-primary"
   712→            placeholder="Mínimo 8 caracteres"
   713→          />
   714→        </div>
   715→        <div className="flex flex-wrap gap-2.5">
   716→          <Button onClick={saveToVault} disabled={busy} className="font-bold">
   717→            <Upload className="mr-2 h-4 w-4" aria-hidden /> Guardar copia cifrada
   718→          </Button>
   719→        </div>
   720→
   721→        {lastCode && (
   722→          <div className="anim-pop rounded-2xl border border-success/50 bg-success/10 p-3">
   723→            <p className="text-xs font-bold text-success">Código de tu sobre (anótalo):</p>
   724→            <p className="mt-1 break-all rounded-lg bg-card px-2 py-1.5 font-mono text-[11px]">{lastCode}</p>
   725→            <p className="mt-1 text-[11px] text-muted-foreground">
   726→              Se borra de la bóveda a los 30 días. Sin la contraseña, nadie (ni nosotros) puede abrirlo.
   727→            </p>
   728→          </div>
   729→        )}
   730→
   731→        <div>
   732→          <Label htmlFor="vault-code" className="text-sm font-bold">Recuperar con código</Label>
   733→          <input
   734→            id="vault-code"
   735→            value={code}
   736→            onChange={(e) => setCode(e.target.value)}
   737→            aria-label="Código del sobre guardado"
   738→            className="mt-1.5 h-10 w-full rounded-2xl border border-border bg-card px-3 font-mono text-xs outline-none focus:border-primary"
   739→            placeholder="Pega aquí el código de 64 caracteres"
   740→          />
   741→        </div>
   742→        <Button variant="outline" onClick={restoreFromVault} disabled={busy} className="font-bold">
   743→          <Download className="mr-2 h-4 w-4" aria-hidden /> Abrir sobre y restaurar
   744→        </Button>
   745→      </div>
   746→
   747→      <p className="mt-3 text-[11px] leading-relaxed text-muted-foreground">
   748→        El servidor no guarda contraseñas, correos ni cookies de sesión: solo sobres cifrados
   749→        que caducan solos. Para máxima privacidad, sigue disponible la exportación manual de JSON.
   750→      </p>
   751→    </section>
   752→  );
   753→}
